	\begin{center}

		{\LARGE Bodies in motion} \bigbreak
		Time to reach speed \smallbreak
		$ t = \frac{v - u}{a} $ \bigbreak

	\end{center}

	Where {\it t} is time (seconds), {\it v} is target speed ({\it m/s}), {\it u} is initial speed ({\it m/s}), and {\it a} is acceleration in ($ m/s^2 $). \bigbreak
	\bigbreak

	\begin{center}

		Power-to-acceleration \smallbreak
		$ a = \sqrt{\frac {P}{2mt}} $ \bigbreak

	\end{center}

	Where {\it a} is acceleration in $ m/s^2 $, {\it P} is power in Watts, {\it m} is mass in kilograms, and {\it t} is time in seconds. \bigbreak
	\bigbreak

	\begin{center}

		Object falling from rest \medbreak

		Conservation of energy \smallbreak
		$ mgh = \frac{1}{2} mv^2 $ \bigbreak

	\end{center}

	Where {\it m} is mass, {\it g} is the gravitational constant ($ 9.80665m/s^2 $ for Earth), {\it h} is height, and {\it v} is velocity in {\it m/s} \bigbreak
	\bigbreak

	\begin{center}

		Kinetic energy \smallbreak
		$ E_J = \frac{1}{2} mv^2 $ \smallbreak

	\end{center}

	Where $ E_J $ is energy in Joules, {\it m} is mass in kilograms, and v is velocity in {\it m/s}.
